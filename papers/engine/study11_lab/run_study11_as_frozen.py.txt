def u_treat(pE): return 1 - 1.2 * pE                       # E[u] of "treat"; "nothing" has 0


def exact(cond):
    w, a1, a2 = COND[cond]; P = law(w, a1, a2); mu, M, mS = moments(P); v = fit_v(mu, M)
    vt = np.sqrt(w * (1 - w)) * (np.array(a1) - np.array(a2))
    if np.dot(v, vt) < 0: v = -v
    return P, mu, M, mS, v, vt


def seg_curve(mu, v, event, k=81):
    lo, hi = segment(mu, v); ts = np.exp(np.linspace(np.log(lo), np.log(hi), k))
    return ts, np.array([event_prob(law(*explanation(mu, v, t)), event) for t in ts])


def voi_exact(mu, v, S, event=E3):
    if abs(np.prod([v[i] for i in S])) < 1e-12: return 0.0
    ts, ps = seg_curve(mu, v, event)
    before = max(0.0, u_treat(ps.max()))                    # best act's worst case over the segment
    after_worst = min(max(0.0, u_treat(p)) for p in ps)     # adversary picks the truth, the probe reveals it
    return float(after_worst - before)


def random_truth(rng, m=4):
    while True:
        w = rng.uniform(.15, .45); a1 = rng.uniform(.1, .9, m); a2 = rng.uniform(.1, .9, m)
        if np.all(np.abs(a1 - a2) > 0.1): return w, a1, a2
